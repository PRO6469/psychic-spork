name: 编译 EdgeGuard APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:  # 允许手动触发

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      # 第一步：拉取代码
      - name: 拉取代码
        uses: actions/checkout@v4

      # 第二步：设置 JDK 17
      - name: 设置 Java 环境
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'

      # 第三步：给 Gradle 脚本添加执行权限
      - name: 授权 Gradle
        run: chmod +x gradlew

      # 第四步：编译 Debug APK（无需签名，可直接安装）
      - name: 编译 APK
        run: ./gradlew assembleDebug

      # 第五步：上传 APK 为 Artifact（可下载）
      - name: 上传 APK
        uses: actions/upload-artifact@v4
        with:
          name: EdgeGuard-APK
          path: app/build/outputs/apk/debug/app-debug.apk
          retention-days: 30
